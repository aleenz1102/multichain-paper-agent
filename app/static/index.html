function drawCandles(candles){
 const svg=$('detailChart');if(!candles||!candles.length){svg.innerHTML='No candle data available.';return}
 const W=900,H=330,L=58,R=18,T=16,B=28,innerW=W-L-R,innerH=H-T-B;
 const lo=Math.min(...candles.map(c=>c.low)),hi=Math.max(...candles.map(c=>c.high)),pad=(hi-lo)*.08||hi*.01,min=lo-pad,max=hi+pad;
 const y=v=>T+(max-v)/(max-min)*innerH,step=innerW/candles.length,body=Math.max(2,Math.min(8,step*.62));
 let out='';
 for(let k=0;k<=4;k++){const v=min+(max-min)*k/4,yy=y(v);out+=''+num(v,6)+''}
 candles.forEach((c,i)=>{const x=L+i*step+step/2,up=c.close>=c.open,cls=up?'up':'down',yo=y(c.open),yc=y(c.close),top=Math.min(yo,yc),height=Math.max(1,Math.abs(yc-yo));out+=''});
 out+=''+new Date(candles[0].timestamp).toLocaleString()+''+new Date(candles[candles.length-1].timestamp).toLocaleString()+'';
 svg.innerHTML=out;
}
